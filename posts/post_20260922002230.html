一、工业现场真实硬件环境与底层痛点深度剖析

"不是所有的工厂都有稳定的网络。"这句话我在上海闵行某汽车零部件厂的车间里说过不下十遍。

1.1 现场硬件环境的"三重地狱"

我们团队在华东地区落地了超过200套工业边缘采集网关，涵盖CNC机床、PLC控制系统、温湿度传感器阵列、AGV小车调度系统等多个场景。现场硬件环境的复杂性远超实验室想象：

串口层的反射与抖动问题

RS-485总线在工业现场是最常见的物理层协议，但它的半双工特性导致了一系列底层痛点。以某压铸车间的20台模温机为例，每台设备通过RS-485总线以Modbus RTU协议上报数据，波特率9600，单帧最大32字节。当多台设备轮询时，总线上会出现明显的信号反射——尤其是当485总线未做终端匹配电阻（120Ω）处理时，信号在总线末端反射回源端，造成数据位跳变。

实测数据显示，未加匹配电阻时，误码率可达$10^{-3}$级别；加120Ω终端电阻后，误码率降至$10^{-6}$以下。但实际生产中，工人经常插拔设备，终端电阻被断开是常态，这意味着网关必须具备强大的数据校验与纠错能力。

网络层的断连与延迟抖动

工厂内部的无线网络环境极其恶劣。某电子元器件厂的SMT产线使用了5GHz频段Wi-Fi，但在贴片机（Pick and Place Machine）高速运行时，其电磁干扰导致WiFi信号波动高达±15dBm，丢包率峰值达到12%。更有甚者，某些车间存在大型变频电机，启停时产生的电磁脉冲会通过电源线传导，导致网关的以太网接口出现瞬断。

我们记录了某次真实故障：一条玻璃基板生产线的MES数据采集网关，在凌晨2点突然断网，持续了47分钟。事后排查发现，是该区域的大型UPS在切换备用电池时产生了浪涌，损坏了网关的PHY芯片。这种场景下，如果网关没有本地缓存机制，47分钟内所有时序数据将永久丢失。

时钟同步的漂移陷阱

边缘网关通常依靠RTC（实时时钟）进行时间戳标记，但廉价 RTC 晶振的温度漂移系数约为±20ppm，在-10°C~50°C的车间环境温度下，一天的时钟偏差可达±1.7秒。对于需要按时间序列对齐的多源数据融合场景（如将振动传感器的10kHz采样数据与温度传感器的1Hz数据对齐），这种偏差足以导致数据关联错误。

我们采用的解决方案是：网关启动时通过NTP同步一次，之后启用硬件看门狗定期校准，并结合本地事件的相对时间戳进行补偿——即使NTP不可用，本地数据的时间序列依然保持相对有序。

架构关键设计决策说明：

  • 多级缓存架构：内存环形队列（L1）→ SQLite WAL模式持久化（L2）→ Flash SSD（L3备份），三级缓存确保极端场景下数据不丢失。
  • SQLite WAL模式：Write-Ahead Logging模式允许读写并发，写入时不阻塞读取，特别适合网关这种"高频写入、低频读取"的场景。
  • 断网检测状态机：采用指数退避的心跳探测机制，从每秒1次逐步降至每30秒1次，节省资源的同时快速响应网络恢复事件。
  • 数据完整性校验：每条记录携带CRC32校验码和顺序号，云端接收后验证，缺失或乱序数据自动触发补传。

3.1 SQLite时序数据持久化存储层

package cache

import (
    "database/sql"
    "fmt"
    "log"
    "sync"
    "time"

    // SQLite驱动，启用WAL模式
    _ "github.com/mattn/go-sqlite3"
)

// TimeSeriesRecord 时序数据记录结构
type TimeSeriesRecord struct {
    ID         uint64    `json:"id"`          // 自增主键
    Timestamp  time.Time `json:"timestamp"`   // 设备上报时间
    DeviceID   string    `json:"device_id"`   // 设备唯一标识
    MetricName string    `json:"metric_name"` // 指标名称
    Value      float64   `json:"value"`       // 指标值
    CRC32      string    `json:"crc32"`       // CRC32校验码（防篡改）
    Buffered   bool      `json:"buffered"`    // 是否已发送确认
}

// SQLiteCache 基于SQLite的本地缓存管理器
type SQLiteCache struct {
    db       *sql.DB
    mu       sync.Mutex
    bufSize  int           // 内存缓冲区大小
    flushCh  chan []Record // 缓冲区channel
}

// NewSQLiteCache 初始化SQLite缓存，启用WAL模式
func NewSQLiteCache(dbPath string, bufSize int) (*SQLiteCache, error) {
    // 连接字符串中设置WAL模式和journal mode
    connStr := fmt.Sprintf("%s?_journal=WAL&_foreign_keys=on&_txlock=immediate", dbPath)
    db, err := sql.Open("sqlite3", connStr)
    if err != nil {
        return nil, fmt.Errorf("打开SQLite数据库失败: %w", err)
    }

    // 关键优化：降低写入延迟
    // synchronous=NORMAL: 在每2个事务提交一次fsync，平衡性能和安全性
    // cache_size=-64000: 64MB共享缓存
    // temp_store=MEMORY: 临时表存储在内存中
    _, err = db.Exec(`
        PRAGMA synchronous = NORMAL;
        PRAGMA cache_size = -64000;
        PRAGMA temp_store = MEMORY;
        PRAGMA journal_mode = WAL;
        PRAGMA busy_timeout = 5000;
    `)
    if err != nil {
        return nil, fmt.Errorf("SQLite优化配置失败: %w", err)
    }

    // 创建时序数据表，使用整数时间戳加速查询
    _, err = db.Exec(`
        CREATE TABLE IF NOT EXISTS telemetry_data (
            id            INTEGER PRIMARY KEY AUTOINCREMENT,
            timestamp     INTEGER NOT NULL,      -- Unix时间戳(秒)
            device_id     TEXT NOT NULL,
            metric_name   TEXT NOT NULL,
            value         REAL NOT NULL,
            crc32         TEXT NOT NULL,         -- CRC32校验码
            buffered      INTEGER DEFAULT 0,     -- 0:待发送 1:已确认
            created_at    INTEGER DEFAULT (strftime('%s','now'))
        );
        -- 创建索引：加速按设备+时间的范围查询和待发送记录筛选
        CREATE INDEX IF NOT EXISTS idx_telemetry_device_ts ON telemetry_data(device_id, timestamp);
        CREATE INDEX IF NOT EXISTS idx_telemetry_unbuffered ON telemetry_data(buffered, id);
    `)
    if err != nil {
        return nil, fmt.Errorf("创建数据表失败: %w", err)
    }

    cache := &SQLiteCache{
        db:       db,
        bufSize:  bufSize,
        flushCh:  make(chan []Record, bufSize),
    }

    // 启动后台刷写协程
    go cache.flushLoop()

    return cache, nil
}

// Save 保存一条时序数据到SQLite
func (c *SQLiteCache) Save(r Record) error {
    c.mu.Lock()
    defer c.mu.Unlock()

    // 转换为Unix时间戳存储，比TEXT格式查询快3倍
    ts := r.Timestamp.Unix()

    // INSERT OR IGNORE防止重复插入（基于唯一约束device_id+timestamp+metric_name）
    _, err := c.db.Exec(
        `INSERT OR IGNORE INTO telemetry_data 
         (timestamp, device_id, metric_name, value, crc32, buffered)
         VALUES (?, ?, ?, ?, ?, 0)`,
        ts, r.DeviceID, r.MetricName, r.Value, r.CRC32,
    )
    if err != nil {
        return fmt.Errorf("写入SQLite失败: %w", err)
    }

    // 通知后台刷写协程处理
    select {
    case c.flushCh <- r:
    default:
        // 通道满时静默丢弃，由后台周期性批量刷写兜底
    }

    return nil
}

// GetPendingRecords 获取待发送的记录（按时间顺序）
func (c *SQLiteCache) GetPendingRecords(limit int) ([]Record, error) {
    c.mu.Lock()
    defer c.mu.Unlock()

    rows, err := c.db.Query(
        `SELECT timestamp, device_id, metric_name, value, crc32 
         FROM telemetry_data 
         WHERE buffered = 0 
         ORDER BY timestamp ASC, id ASC 
         LIMIT ?`,
        limit,
    )
    if err != nil {
        return nil, fmt.Errorf("查询待发送记录失败: %w", err)
    }
    defer rows.Close()

    var records []Record
    for rows.Next() {
        var r Record
        var ts int64
        if err := rows.Scan(&ts, &r.DeviceID, &r.MetricName, &r.Value, &r.CRC32); err != nil {
            return nil, err
        }
        r.Timestamp = time.Unix(ts, 0)
        records = append(records, r)
    }
    return records, rows.Err()
}

// MarkAsBuffered 标记记录为已发送（云端确认回执后调用）
func (c *SQLiteCache) MarkAsBuffered(ids []uint64) error {
    c.mu.Lock()
    defer c.mu.Unlock()

    if len(ids) == 0 {
        return nil
    }

    // 批量更新，避免N+1查询问题
    placeholders := make([]string, len(ids))
    args := make([]interface{}, len(ids)+1)
    for i, id := range ids {
        placeholders[i] = "?"
        args[i] = id
    }
    args[len(ids)] = 1 // buffered = 1

    sql := fmt.Sprintf(
        `UPDATE telemetry_data SET buffered = 1 WHERE id IN (%s) AND buffered = 0`,
        strings.Join(placeholders, ","),
    )
    _, err := c.db.Exec(sql, args...)
    return err
}

// Statistics 获取缓存统计信息（用于监控看板）
func (c *SQLiteCache) Statistics() (total, pending int64, err error) {
    err = c.db.QueryRow("SELECT COUNT(*) FROM telemetry_data").Scan(&total)
    if err != nil {
        return 0, 0, err
    }
    err = c.db.QueryRow("SELECT COUNT(*) FROM telemetry_data WHERE buffered = 0").Scan(&pending)
    return total, pending, err
}
