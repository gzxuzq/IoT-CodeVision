1. 项目背景与核心挑战

1.1 行业痛点

现代集装箱码头与电商前置仓面临三重叠加压力：

维度 现状痛点 期望目标
吞吐效率 人工驾驶集卡/叉车峰值响应 3–5 min，排队拥堵严重 AGV 响应 < 200 ms，队列零等待
通信可靠性 Wi-Fi 6 在金属环境多径衰落，丢包率 2–5% 导致急停 5G 私网确定性时延 < 10 ms，可用性 99.999%
调度扩展性 集中式调度器单点瓶颈，50+ 台车后性能断崖 水平扩展至 500+ 台，调度延迟线性增长

1.2 场景定义

场景A：自动化集装箱码头（OCT — Open Container Terminal）
  └─ 岸桥 ↔ 场桥 ↔ 跨运车 ↔ 集卡 之间的无人化物料流转

场景B：电商立体仓库（AS/RS + AMR 混合）
  └─ 立库堆垛机 ↔ AGV 接驳 ↔ 分拣线 ↔ 发货区 的柔性协同

1.3 核心设计约束

  • 协议栈：5G 专网（传输层）+ MQTT 5.0（设备消息）+ WebSocket（实时大屏/管控）+ gRPC（服务间 RPC）
  • 并发规模：500 台 AGV × 20 消息/秒 = 10,000 TPS 上行；调度指令 500 TPS 下行
  • 时延 SLA：感知→决策→执行全链路 ≤ 50 ms（P99）
  • 可用性：99.99%（年停机 ≤ 53 min）

3. 通信协议栈设计

3.1 协议分工矩阵

层级 协议 用途 QoS / SLA
设备层 MQTT 5.0 AGV 状态上报 / 订阅指令 QoS 1，Retain=false，Properties 携带 DeviceID
实时层 WebSocket 5.0 操作员控制台 / 数字孪生渲染 持久连接，心跳 5s，二进制帧
服务层 gRPC / Protobuf 调度器 ↔ 网关 / 网关 ↔ 上游系统 HTTP/2 多路复用， deadline 100ms
传输层 5G URLLC + 本地 UPF 端到端承载，零公网穿越 QCI 1，时延抖动 < 2 ms

3.2 MQTT 5.0 设计细节

Topic 架构（层级化 + 通配符友好）：

 港口/agv/{zone}/{vid}/telemetry     ← 上行：状态遥测（20 Hz）
 港口/agv/{zone}/{vid}/command       ← 下行：控制指令（<= 10 Hz）
 港口/agv/{zone}/{vid}/fault         ← 告警事件（事件驱动）
 港口/routing/{zone}/tasks           ← 全局任务广播（QoS 0）
 港口/fleet/{zone}/health            ← 车队健康（聚合，QoS 1）

MQTT 5.0 特性启用：
  - Shared Subscription: 多订阅者负载均衡（$share/group/topic）
  - Reason Code: 精确错误语义（CONTINUE / UNSATISFIED_QOS / REFUSED）
  - User Properties: 携带扩展字段（seqNo, timestamp, priority）
  - Topic Alias: 减少带宽占用（长 topic 用 2 字节别名替换）
  - Message Expiry: 指令 TTL 10s，过期自动丢弃
  - Payload Format Indicator: 二进制备注 Protobuf 序列化

3.3 Protobuf 消息定义（核心）

// agv_telemetry.proto
message TelemetryPacket {
  uint64  device_id      = 1;  // AGV UUID
  int64   timestamp_us   = 2;  // 微秒级 UTC
  float32 x_m            = 3;  // 位置 X
  float32 y_m            = 4;  // 位置 Y
  float32 heading_deg    = 5;  // 航向角
  float32 velocity_ms    = 6;  // 当前速度
  float32 battery_pct    = 7;  // 电量百分比
  int32   state          = 8;  // IDLE/RUNNING/CHARGING/FAULT
  repeated Obstacle nearby_obstacles = 9;
}

// command.proto
message AgvCommand {
  uint64 device_id    = 1;
  int64  seq_no       = 2;
  int64  deadline_us  = 3;  // 必须在 deadline 前执行
  oneof action {
    Waypoint waypoint = 10;
    SpeedLimit speed_limit = 11;
    EmergencyStop emergency_stop = 12;
    ChargeStart charge_start = 13;
  }
}

message Waypoint {
  float32 x_m = 1;
  float32 y_m = 2;
  float32 heading_deg = 3;
  float32 speed_ms = 4;
  float32 approach_radius_m = 5;  // 接近判定半径
}

3.4 gRPC 服务定义

// dispatch.proto
service DispatchService {
  // 批量派发任务给指定 AGV
  rpc AssignTasks (TaskBatchRequest) returns (TaskBatchResponse);
  // 实时查询路径与 ETA
  rpc QueryPath (PathQueryRequest) returns (stream PathUpdate);
  // 车对车协商通行权（分布式互锁）
  rpc NegotiateRightOfWay (ROWNegotiationRequest)
      returns (ROWNegotiationResponse);
  // 区域健康检查与接管
  rpc ZoneHealthCheck (ZoneCheckRequest) returns (ZoneHealthResponse);
}

// 区域协调服务（Zone Coordinator）
service ZoneCoordinationService {
  rpc ReserveSpace (SpaceReservationRequest) returns (SpaceReservationResponse);
  rpc ReleaseSpace (SpaceReleaseRequest) returns (SpaceReleaseResponse);
  rpc BroadcastEvent (ZoneEvent) returns (google.protobuf.Empty);
}
