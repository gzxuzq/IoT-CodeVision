一、工业现场真实硬件环境与底层痛点深度剖析

1.1 产线环境画像：不是理想实验室，是电磁风暴中心

在汽车零部件焊接车间部署边缘采集节点时，我们面对的是这样的现场：

  • 环境温度：-10°C ~ 65°C（靠近焊接机器人热辐射区）
  • 电磁干扰：多台6KVA伺服焊机同时启动，总线噪声峰值达到±15V
  • 机械振动：地面传导振动频率2~80Hz，端子松动失效是高频故障点
  • 供电波动：485总线供电与PLC共用电源轨，浪涌冲击频繁

1.2 Modbus RTU 协议栈的先天性缺陷

1.2.1 报文时序的脆弱性

Modbus RTU 依赖于精确的字符间隔和帧间隔：

T3.5 字符间隔（静止期）→ 开始传输 → T1.5/T3.5 帧间隔

实际观测到的问题：

现象 根因 影响
报文截断 波特率 19200 下 T3.5 ≈ 1.8ms，UART FIFO 溢出 CRC 校验失败率 > 8%
粘包 DMA 接收缓冲区未清空即触发下一次轮询 帧头错位，全量数据污染
反射回波 485 总线阻抗不匹配（120Ω vs 实际 95Ω） 上升沿过冲，逻辑电平误判

1.2.2 RS-485 总线的抖动陷阱

半双工总线的先天冲突：多节点共享总线，主站切换方向时存在切换延迟（通常 50~200μs）。在我们的实测中：

正常场景：主站发送请求 → 等待 T3.5 → 从站响应
异常场景：主站切换为接收模式过快 → 从站应答被截断

解决方案核心思路：必须在驱动层实现自适应忙检测，而非固定延时。

1.2.3 网络断连的级联效应

当 MQTTS 断连超过 30s，传统做法是直接丢弃缓存，导致： - SCADA 系统视角出现"数据空洞" - 生产追溯系统无法关联时间窗口 - 事后故障定位缺失关键时序数据

logger = logging.getLogger("modbus_adapter")

class BusState(Enum): IDLE = "idle" POLLING = "polling" WAITING = "waiting" ERROR = "error"

@dataclass class ModbusFrame: """Modbus RTU 帧结构""" slave_id: int function_code: int start_addr: int quantity: int raw_bytes: bytes timestamp: float crc_ok: bool

@dataclass class PollConfig: """轮询配置""" slave_ids: List[int] register_map: Dict[int, Dict[str, any]] poll_interval_ms: int = 500 retry_count: int = 3 timeout_ms: int = 200

核心设计思想：
1. 动态计算 T3.5 等效等待时间（而非固定延时）
2. 基于 CRC 错误率自适应调整波特率和超时参数
3. 方向切换采用忙检测，避免固定延迟导致的帧丢失

    57600: 0.173e-3,
    115200: 0.087e-3,
}

def __init__(self, config: PollConfig, serial_port: str = "/dev/ttyUSB0"):
    self.config = config
    self.serial_port = serial_port
    self._ser: Optional[serial.Serial] = None
    self._bus_state = BusState.IDLE
    self._direction_lock = asyncio.Lock()
    self._stats = {
        "total_polls": 0,
        "crc_errors": 0,
        "timeout_errors": 0,
        "retry_count": 0,
        "bytes_transmitted": 0,
        "bytes_received": 0,
    }

    # 自适应参数
    self._error_rate_window = []

    self._on_frame: Optional[Callable[[ModbusFrame], None]] = None
    self._on_data: Optional[Callable[[Dict], None]] = None

def setup(self) -> bool:
    """初始化串口，启用硬件流控"""
    try:
        # 关键：启用 RTS/CTS 硬件流控，避免方向切换冲突
        self._ser = serial.Serial(
            port=self.serial_port,
            baudrate=19200,
            bytesize=serial.EIGHTBITS,
            parity=serial.PARITY_NONE,
            stopbits=serial.STOPBITS_1,
            timeout=0.1,  # 非阻塞读超时
            rtscts=True,  # 硬件流控，自动管理 DE/RE 引脚
            dsrdtr=True,
        )

        # 优化：设置读写缓冲区大小
        self._ser.read_buf_size = 4096
        self._ser.write_buf_size = 4096

        logger.info(f"串口 {self.serial_port} 初始化成功")
        return True

    except serial.SerialException as e:
        logger.error(f"串口初始化失败: {e}")
        return False

def compute_t35(self, baudrate: int = 19200) -> float:
    """
    计算 T3.5 理论时间（秒）

    Modbus 规范：T3.5 = 3.5 × 字符时长
    字符时长 = 11 bits / baudrate（含1起始位+8数据位+1校验位+1停止位）
    """
    bits_per_char = 11
    char_duration = bits_per_char / baudrate
    t35 = 3.5 * char_duration
    return t35

async def _wait_for_bus_idle(self, min_wait_ms: float = 5.0):
    """
    等待总线空闲

    使用忙检测：监测 RTS 引脚状态，确保上一次发送完全结束
    """
    if self._ser and self._ser.is_open:
        # 等待至少一个字符时长，确保方向切换稳定
        await asyncio.sleep(min_wait_ms / 1000.0)
        # 额外等待 T3.5 时间
        t35 = self.compute_t35()
        await asyncio.sleep(t35)

async def _send_request(self, slave_id: int, function_code: int, 
                       start_addr: int, quantity: int) -> bytes:
    """
    构建并发送 Modbus RTU 请求帧

    帧格式: [地址][功能码][起始地址高][起始地址低][数量高][数量低][CRC低][CRC高]
    """
    # 构建请求数据（不含 CRC）
    raw_cmd = struct.pack(">BBHH", 
                         slave_id,
                         function_code,
                         start_addr,
                         quantity)

    # 计算 CRC16（Modbus 算法）
    crc = self._calc_crc16(raw_cmd)
    frame = raw_cmd + struct.pack("<H", crc)

    # 切换方向为发送模式
    async with self._direction_lock:
        if self._ser and self._ser.is_open:
            self._ser.write(frame)
            self._stats["bytes_transmitted"] += len(frame)

    return frame

def _calc_crc16(self, data: bytes) -> int:
    """标准 Modbus CRC16 计算"""
    crc = 0xFFFF
    for byte in data:
        crc ^= byte
        for _ in range(8):
            if crc & 0x0001:
                crc >>= 1
                crc ^= 0xA001  # Modbus 多项式
            else:
                crc >>= 1
    return crc

async def _read_response(self, expected_len: int = 0, 
                         timeout_ms: int = 200) -> Optional[bytes]:
    """
    读取从站响应，带超时和重试

    关键：使用 select 超时而非阻塞读，避免单节点故障拖垮整条总线
    """
    if not self._ser or not self._ser.is_open:
        return None

    loop = asyncio.get_event_loop()
    try:
        # 使用 asyncio 包装串行读取
        read_task = loop.run_in_executor(
            None,
            lambda: self._ser.read(expected_len or 256)
        )

        # 设置超时
        try:
            response = await asyncio.wait_for(read_task, timeout=timeout_ms/1000.0)
        except asyncio.TimeoutError:
            self._stats["timeout_errors"] += 1
            logger.warning(f"读取超时，timeout={timeout_ms}ms")
            return None

        if response:
            self._stats["bytes_received"] += len(response)
        return response

    except Exception as e:
        logger.error(f"读取响应异常: {e}")
        return None

def _verify_crc(self, frame: bytes) -> bool:
    """验证帧 CRC"""

        return False

    return self._calc_crc16(data) == received_crc
