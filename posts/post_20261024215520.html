作者：码视野物联网软件研发团队 · 首席资深架构师 领域：工业 IoT / 边缘计算 · 深耕 6 年

一、工业现场真实硬件环境与底层痛点深度剖析

我负责的产线数据采集项目，部署在一家新能源汽车电池制造工厂。现场环境恶劣：高温、高湿、强电磁干扰（MES系统、焊机、机械臂同时运行），PLC 通过 RS-485 总线挂载了 32 台变频器，每条总线最远通信距离 800 米，上行网络依赖千兆工业以太网，但日均断网次数 3~7 次，每次持续 30 秒至 15 分钟不等。

1.1 现场真实痛点

① 报文时序错乱（时间戳漂移）

Modbus TCP 轮询周期 200ms，但由于网关 Linux 内核的 jiffies 计时在 CPU 高负载时存在毫秒级抖动，加上网络 RTT 波动（5ms~200ms），原始采集数据的时间戳序列经常出现乱序或重复。若直接入库，后续数据分析完全失真。

② 串口反射（Serial Port Reflection）

RS-485 半双工模式下，从设备响应超时后，主站未正确释放总线，导致下一条报文被"反射"到错误设备，整条总线僵死。重启 485 收发器芯片（gpioctl 拉低复位）是现场工程师唯一的临时恢复手段。

③ 485 总线抖动（Noise Coupling）

焊机启动瞬间，总线上出现 5kHz~50kHz 的高频噪声，波特率 115200 下误码率飙升至 1e-4。网关驱动层必须实现** CRC 校验 + 自动重读 + 指数退避**，否则数据丢包率高达 12%。

④ 网络断连后的数据黑洞

这是最致命的痛点。断网期间，传感器仍在高频采样（10Hz~100Hz），传统方案要么直接丢弃，要么缓存在 RAM 中。RAM 掉电即失，一旦遭遇市电波动（工厂常见），整条产线的工艺参数数据全部丢失，追溯失效，导致质量事故。

二、码视野高可用边缘系统拓扑架构

2.1 架构核心设计理念

① SQLite WAL 模式保障并发读写

传统 SQLite 写操作会阻塞读操作。我们开启 WAL（Write-Ahead Logging）模式，允许读写并发，采样线程持续写入的同时，续传线程可以实时读取，互不阻塞。

② 时间同步统一时钟源

所有数据写入前，时间戳以NTP 服务器校准后的系统时间为准，避免客户端本地时钟漂移。同时为每条数据打上采集原始时间戳和网关写入时间戳两列，便于事后追溯。

③ 分区表设计

按小时分区（PARTITION BY RANGE），每个分区独立文件，便于定期清理过期数据，同时减少单表索引膨胀。

3.1 数据模型定义（Go struct + SQLite 建表）

package main

import (
    "database/sql"
    "fmt"
    "log"
    "time"

    _ "github.com/mattn/go-sqlite3" // CGO 编译，支持 SQLite
)

// SensorReading 单条传感数据
type SensorReading struct {
    DeviceID     string    // 设备唯一标识
    Tags         string    // Modbus 寄存器地址
    Value        float64   // 采集数值
    SampleTime   time.Time // 传感器原始时间戳（来自 PLC）
    RecvTime     time.Time // 网关接收时间（系统时钟）
    Retried      bool      // 是否已重传
    Checksum     string    // CRC 校验值
}

// InitDatabase 初始化 SQLite 数据库，使用 WAL 模式
func InitDatabase(dbPath string) (*sql.DB, error) {
    db, err := sql.Open("sqlite3", dbPath+"?_journal_mode=WAL&_foreign_keys=on")
    if err != nil {
        return nil, fmt.Errorf("open sqlite: %w", err)
    }

    // 设置连接池，避免并发瓶颈
    db.SetMaxOpenConns(4)
    db.SetMaxIdleConns(2)

    // 创建分区表：按小时分区，减少单表数据量
    createTableSQL := `
    CREATE TABLE IF NOT EXISTS sensor_data (
        id          INTEGER PRIMARY KEY AUTOINCREMENT,
        device_id   TEXT NOT NULL,
        tags        TEXT NOT NULL,
        value       REAL NOT NULL,
        sample_time INTEGER NOT NULL,
        recv_time   INTEGER NOT NULL DEFAULT (strftime('%s','now')),
        retried     INTEGER NOT NULL DEFAULT 0,
        checksum    TEXT,
        index idx_recv_time (recv_time),
        index idx_device_sample (device_id, sample_time)
    );`

    if _, err := db.Exec(createTableSQL); err != nil {
        return nil, fmt.Errorf("create table: %w", err)
    }

    // 创建续传队列索引视图（仅查询未重传数据）
    viewSQL := `
    CREATE VIEW IF NOT EXISTS pending_retry AS
    SELECT * FROM sensor_data WHERE retried = 0;`

    if _, err := db.Exec(viewSQL); err != nil {
        return nil, fmt.Errorf("create view: %w", err)
    }

    log.Printf("[DB] SQLite initialized with WAL mode: %s", dbPath)
    return db, nil
}

3.2 采集写入层（采集线程 + 批量写入）

// BatchWriter 批量写入缓冲器，减少 SQLite 事务开销
type BatchWriter struct {
    db      *sql.DB
    batch   []SensorReading
    batchSize int
    flush   chan struct{} // 触发刷盘
}

func NewBatchWriter(db *sql.DB, batchSize int) *BatchWriter {
    return &BatchWriter{
        db:        db,
        batchSize: batchSize,
        flush:     make(chan struct{}, 1),
    }
}

// Add 加入采集数据
func (bw *BatchWriter) Add(r SensorReading) {
    bw.batch = append(bw.batch, r)
    if len(bw.batch) >= bw.batchSize {
        bw.flush <- struct{}{}
    }
}

// Flush 批量写入 SQLite
func (bw *BatchWriter) Flush() {
    if len(bw.batch) == 0 {
        return
    }

    tx, err := bw.db.Begin()
    if err != nil {
        log.Printf("[WRITE] begin tx error: %v", err)
        return
    }

    stmt, err := tx.Prepare(`
        INSERT INTO sensor_data 
        (device_id, tags, value, sample_time, recv_time, checksum)
        VALUES (?, ?, ?, ?, ?, ?)
    `)
    if err != nil {
        tx.Rollback()
        log.Printf("[WRITE] prepare error: %v", err)
        return
    }
    defer stmt.Close()

    for _, r := range bw.batch {
        sampleTS := r.SampleTime.Unix()
        recvTS := r.RecvTime.Unix()
        if _, err := stmt.Exec(r.DeviceID, r.Tags, r.Value,
            sampleTS, recvTS, r.Checksum); err != nil {
            log.Printf("[WRITE] insert error: %v", err)
            tx.Rollback()
            return
        }
    }

    if err := tx.Commit(); err != nil {
        log.Printf("[WRITE] commit error: %v", err)
        return
    }

    log.Printf("[WRITE] flushed %d records to SQLite", len(bw.batch))
    bw.batch = bw.batch[:0]
}

// Run 启动后台刷盘协程，定时刷新
func (bw *BatchWriter) Run() {
    ticker := time.NewTicker(500 * time.Millisecond)
    defer ticker.Stop()

    for {
        select {
        case <-ticker.C:
            bw.Flush()
        case <-bw.flush:
            bw.Flush()
        }
    }
}
