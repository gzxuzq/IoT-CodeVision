一、工业现场真实硬件环境与底层痛点

1.1 产线传感器生态的复杂性

站在某汽车零部件焊接车间的产线旁，你面对的是这样的硬件生态：

  • 128台KUKA焊接机器人：每台输出CSV格式的焊缝参数（电流、电压、速度、温度），采样频率200Hz
  • 64台ABB视觉检测系统：每帧图像约2MB，每秒产生4-8个检测结果
  • 32套三菱PLC：Modbus TCP协议，500ms周期轮询
  • 8台AGV小车：MQTT over TCP，实时位置上报
  • 温湿度、振动、噪声传感器：NB-IoT接入，分钟级上报

总设备数：232台，日均产生数据量约4.7TB，峰值QPS达3,200+

1.2 报文时序问题：工业时序的本质挑战

在MES系统中，我们曾遇到一个典型问题：焊接参数与检测结果无法对齐。

时间轴（μs）：
| Robot-A |  PLC-01 | Vision-C |
| 200ms   | 500ms   | 100ms    |
| 异步    | 异步    | 异步     |

三个子系统时钟不同步，导致数据关联准确率只有73%。

根因分析： 1. 没有统一的NTP服务：各设备独立时钟 2. 协议差异大：Modbus、CANopen、Profinet、MQTT混用 3. 边缘网关处理能力不足：老旧网关CPU占用率85%+

1.3 串口反射与485总线抖动

某铸造厂的生产环境：

RS-485总线拓扑：
[PLC-01]---/\/\/\---[仪表-02]---/\/\/\---[仪表-03]---GND
              120Ω                 120Ω

信号反射现象：
- 波特率9600时误码率 0.001%
- 波特率19200时误码率 0.05%  ← 不可接受
- 长距离（300m）时数据丢失率 12%

工程解决方案： 1. 终端匹配电阻（120Ω）必须加装 2. 波特率降级到9600，增加CRC校验 3. 边缘侧加FIFO缓冲，实现断线重传

1.4 网络断连与数据持久化

真实事故案例：

2023年11月，某钢铁厂MES系统宕机3小时，导致： - 23,000条工艺参数丢失 - 8小时生产记录无法追溯 - 经济损失约120万元

根因：

原架构：
传感器 → 网关 → 直连MES数据库
           ↓
        无缓冲、无重试、无持久化

3.1 Python版：MQTT+Kafka双通道数据管道

"""
码视野工业遥测数据管道
功能：MQTT采集 + Kafka持久化 + Redis缓存
环境：Python 3.9+, paho-mqtt, confluent-kafka, redis
"""

import paho.mqtt.client as mqtt
from confluent_kafka import Producer, Consumer, KafkaError
import redis
import json
import time
import threading
from datetime import datetime

class IndustrialDataPipeline:
    """工业级遥测数据管道"""

    def __init__(self, config):
        # MQTT配置（EMQX Broker）
        self.mqtt_broker = config['mqtt']['broker']
        self.mqtt_port = config['mqtt']['port']
        self.mqtt_topic = config['mqtt']['topic']

        # Kafka配置
        self.kafka_brokers = config['kafka']['brokers']
        self.kafka_topic = config['kafka']['topic']

        # Redis配置
        self.redis_host = config['redis']['host']
        self.redis_port = config['redis']['port']

        # 初始化组件
        self._init_mqtt()
        self._init_kafka()
        self._init_redis()

        # 统计指标
        self.stats = {
            'mqtt_received': 0,
            'kafka_produced': 0,
            'redis_cached': 0,
            'errors': 0
        }

    def _init_mqtt(self):
        """初始化MQTT客户端（EMQX连接）"""
        self.mqtt_client = mqtt.Client(
            client_id=f"mv_{socket.gethostname()}_{int(time.time())}",
            protocol=mqtt.MQTTv5
        )
        self.mqtt_client.on_connect = self._on_mqtt_connect
        self.mqtt_client.on_message = self._on_mqtt_message
        self.mqtt_client.on_disconnect = self._on_mqtt_disconnect

        # 设置遗嘱消息（LWT）
        self.mqtt_client.will_set(
            topic="devices/lost",
            payload=json.dumps({"status": "offline", "ts": datetime.now().isoformat()}),
            qos=1,
            retain=True
        )

        # 启用TLS（工业环境必备）
        self.mqtt_client.tls_set(
            ca_certs=config['tls']['ca_cert'],
            certfile=config['tls']['client_cert'],
            keyfile=config['tls']['client_key']
        )

    def _init_kafka(self):
        """初始化Kafka生产者"""
        self.kafka_producer = Producer({
            'bootstrap.servers': ','.join(self.kafka_brokers),
            'compression.type': 'lz4',      # 工业数据压缩
            'message.max.bytes': 1048576,    # 1MB消息上限
            'delivery.timeout.ms': 30000,    # 30秒超时
            'linger.ms': 10,                 # 批处理延迟
            'batch.size': 32768               # 批次大小
        })

    def _init_redis(self):
        """初始化Redis缓存"""
        self.redis_client = redis.Redis(
            host=self.redis_host,
            port=self.redis_port,
            db=0,
            decode_responses=True,
            socket_timeout=5,
            socket_connect_timeout=3
        )

    def _on_mqtt_connect(self, client, userdata, flags, rc, properties=None):
        """MQTT连接回调"""
        if rc == 0:
            print(f"[MQTT] 已连接到 EMQX: {self.mqtt_broker}")
            # 订阅工业数据主题
            client.subscribe(self.mqtt_topic, qos=1)
            print(f"[MQTT] 已订阅: {self.mqtt_topic}")
        else:
            print(f"[MQTT] 连接失败: {rc}")

    def _on_mqtt_message(self, client, userdata, msg):
        """MQTT消息处理"""
        try:
            payload = json.loads(msg.payload.decode())
            self.stats['mqtt_received'] += 1

            # 1. 写入Kafka持久化
            kafka_msg = json.dumps({
                **payload,
                'source': 'mqtt',
                'timestamp': datetime.now().isoformat(),
                'topic': msg.topic
            }).encode()

            self.kafka_producer.produce(
                self.kafka_topic,
                key=payload.get('device_id', b'').encode(),
                value=kafka_msg
            )
            self.stats['kafka_produced'] += 1

            # 2. 缓存到Redis（最近1000条）
            redis_key = f"latest:{payload.get('device_id')}"
            self.redis_client.lpush(redis_key, json.dumps(payload))
            self.redis_client.ltrim(redis_key, 0, 999)
            self.stats['redis_cached'] += 1

            # 3. 发布确认消息
            ack = {"status": "received", "device_id": payload.get('device_id')}
            client.publish("devices/ack", json.dumps(ack))

        except Exception as e:
            self.stats['errors'] += 1
            print(f"[ERROR] 消息处理失败: {e}")

    def _on_mqtt_disconnect(self, client, userdata, rc):
        """MQTT断开回调（实现断线重连）"""
        print(f"[MQTT] 断开，尝试重连... ({rc})")
        while True:
            try:
                time.sleep(5)
                client.reconnect()
                print("[MQTT] 重连成功")
                break
            except:
                print("[MQTT] 重连失败，继续重试...")

    def start(self):
        """启动数据管道"""
        # 启动MQTT连接（阻塞）
        self.mqtt_client.connect_async(self.mqtt_broker, self.mqtt_port, keepalive=60)
        self.mqtt_client.loop_start()

        # 定期输出统计
        while True:
            time.sleep(30)
            self._log_stats()

    def _log_stats(self):
        """日志统计"""
        print(f"\n[STATS] MQTT收: {self.stats['mqtt_received']} | "
              f"Kafka发: {self.stats['kafka_produced']} | "
              f"Redis缓存: {self.stats['redis_cached']} | "
              f"错误: {self.stats['errors']}")

# 使用示例
if __name__ == "__main__":
    config = {
        'mqtt': {
            'broker': '192.168.1.100',
            'port': 8883,
            'topic': 'factory/welding/#'
        },
        'kafka': {
            'brokers': ['kafka-01:9092', 'kafka-02:9092', 'kafka-03:9092'],
            'topic': 'industrial.welding.raw'
        },
        'redis': {
            'host': '192.168.1.200',
            'port': 6379
        },
        'tls': {
            'ca_cert': '/etc/mqtt/ca.crt',
            'client_cert': '/etc/mqtt/client.crt',
            'client_key': '/etc/mqtt/client.key'
        }
    }

    pipeline = IndustrialDataPipeline(config)
    pipeline.start()
