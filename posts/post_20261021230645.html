作者：码视野物联网软件研发团队首席架构师 | 工业 IoT 与边缘计算深耕 6 年

一、工业现场真实硬件环境与底层痛点深度剖析

我在产线现场调试过上百台 PLC、CNC 和视觉检测终端，设备上报频率从每秒 1 条到每秒 200 条不等。真实工业现场的底层问题是多态并存、时序错乱、链路脆弱，直接作用于消息中间件选型：

1. 报文时序撕裂
传感器原始数据通过 RS-485/232 汇聚后，同一时间戳多设备并发上送，TCP 粘包 + 应用层协议解析延迟，导致"时间戳乱序 + 事件因果倒置"问题频发。Kafka 的分区幂等写入能解决有序性，但代价是吞吐下降 30%~40%。

2. 串口反射与 485 总线抖动
多台设备共享一条 RS-485 总线时，主站轮询周期 50ms，从站响应时间 10~80ms 漂移。高峰期总线负载率达 85%，CRC 校验错误每百帧出现 3~7 次。边缘网关必须做本地环形缓冲 + 重试窗口对齐，否则直接丢数据。

3. 网络断连与弱网恢复
工厂 Wi-Fi 6 覆盖盲区多，4G 工业路由器切换基站延迟 2~8s。设备离线后本地产生缓存堆积，恢复后若立即全量重传，服务端瞬间压力暴增。背压 + 限速+分区补传是必选项。

结论：纯 Kafka 方案在接入层扛不住设备碎片化；纯 MQTT 方案在持久化与流处理层又不够深。两者必须边界清晰、分层融合。

class LocalCache: """ 断网期间本地落库，使用 SQLite WAL 模式保证写入性能。 表结构：telemetry_raw(id, device_id, timestamp_ms, topic, payload_hex, seq_id, retry_count, status) status: 0=待发送, 1=已发送, 2=丢弃 """

def __init__(self, db_path: str):
    self.db_path = db_path
    self._db: Optional[aiosqlite.Connection] = None
    self._seq_counter = 0

async def init(self):
    self._db = await aiosqlite.connect(self.db_path, timeout=10.0)
    await self._db.execute("PRAGMA journal_mode=WAL")
    await self._db.execute("""
        CREATE TABLE IF NOT EXISTS telemetry_raw (
            id          INTEGER PRIMARY KEY AUTOINCREMENT,
            device_id   TEXT NOT NULL,
            timestamp_ms INTEGER NOT NULL,
            topic       TEXT NOT NULL,
            payload_hex TEXT NOT NULL,
            seq_id      INTEGER NOT NULL,
            retry_count INTEGER DEFAULT 0,
            status      INTEGER DEFAULT 0
        )
    """)
    await self._db.commit()
    # 启动时清理超过 72h 的已发送记录，释放磁盘
    cutoff_ms = int(time.time() * 1000) - 72 * 3600 * 1000
    await self._db.execute(
        "DELETE FROM telemetry_raw WHERE status=1 AND timestamp_ms <?", (cutoff_ms,)
    )
    await self._db.commit()
    logger.info(f"LocalCache initialized: {self.db_path}")

async def save(self, point: TelemetryPoint) -> int:
    """写入本地缓存，返回自增 ID"""
    row_id = await self._db.execute(
        """INSERT INTO telemetry_raw
           (device_id, timestamp_ms, topic, payload_hex, seq_id, retry_count, status)
           VALUES (?, ?, ?, ?, ?, ?, 0)""",
        (point.device_id, point.timestamp_ms, point.topic,
         point.payload.hex(), point.seq_id, point.retry_count)
    )
    await self._db.commit()
    return row_id

async def fetch_pending(self, limit: int = BATCH_FLUSH_SIZE) -> List[TelemetryPoint]:
    """取出待发送记录（status=0），按 seq_id 升序"""
    cursor = await self._db.execute(
        """SELECT device_id, timestamp_ms, topic, payload_hex, retry_count
           FROM telemetry_raw WHERE status=0 ORDER BY seq_id ASC LIMIT ?""",
        (limit,)
    )
    rows = await cursor.fetchall()
    points = []
    for row in rows:
        points.append(TelemetryPoint(
            device_id=row[0],
            timestamp_ms=row[1],
            topic=row[2],
            payload=bytes.fromhex(row[3]),
            seq_id=0,  # seq_id 仅作顺序保证，此处不依赖 DB 字段
            retry_count=row[4]
        ))
    return points

async def mark_sent(self, row_ids: List[int]):
    """批量标记为已发送"""
    placeholders = ",".join("?" * len(row_ids))
    await self._db.execute(
        f"UPDATE telemetry_raw SET status=1 WHERE id IN ({placeholders})",
        row_ids
    )
    await self._db.commit()

async def increment_retry(self, row_id: int):
    await self._db.execute(
        "UPDATE telemetry_raw SET retry_count=retry_count+1 WHERE id=?", (row_id,)
    )
    await self._db.commit()

class MqttPublisher: def init(self, broker: str, port: int, client_id: str, tls: bool = True): self.broker = broker self.port = port self.client_id = client_id self._client = mqtt.Client( client_id=client_id, protocol=mqtt.MQTTv5, clean_start=True, # 持久会话：断线后 MQTTS 服务端保留未 ACK 消息 ) if tls: self._client.tls_set() self._client.on_connect = self._on_connect self._client.on_disconnect = self._on_disconnect self._connected = False

def _on_connect(self, client, userdata, flags, rc, properties=None):
    self._connected = True
    logger.info(f"MQTT connected: rc={rc}, client={self.client_id}")

def _on_disconnect(self, client, userdata, rc, properties=None):
    self._connected = False
    logger.warning(f"MQTT disconnected: rc={rc}")

async def publish(self, topic: str, payload: bytes, qos: int = 1) -> bool:
    if not self._connected:
        return False
    loop = asyncio.get_event_loop()
    return await loop.run_in_executor(
        None,
        lambda: self._client.publish(topic, payload, qos=qos).is_published()
    )
